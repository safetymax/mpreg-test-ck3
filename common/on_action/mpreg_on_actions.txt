##############################################################################
# MPREG - Male Pregnancy -- on_action hooks
# File: common/on_action/mpreg_on_actions.txt
#
# The two blocks below reopen real vanilla on_actions and add to their
# `events = { }` list only. Unlike scripted_triggers/gene templates/portrait
# modifier groups (Lesson 2, Lesson 11), on_action event lists are the one
# database in this project confirmed by Paradox's own documented modding
# convention to be a genuinely additive, cross-file, cross-mod list --
# every other mod that adds a wedding/birth reaction does exactly this.
# Not redeclaring anything else about these on_actions, so nothing vanilla
# already registered on them can be lost.
#
# Lesson 32: on_pregnancy_mother and on_birth_mother used to be hooked here
# too, firing mpreg.0001/mpreg.0003 (the old mixed-sex "redirect" system,
# removed -- see events/mpreg_events.txt). A real vanilla female pregnancy
# needs no hook of any kind anymore; it's entirely untouched, exactly as
# unmodified vanilla, and a Carrier husband's own independent pregnancy runs
# through on_marriage/on_set_relation_lover below like any other couple.
##############################################################################

# common/on_action/marriage_concubinage.txt:4 -- fires after any live (non-historical) marriage.
on_marriage = {
	events = {
		mpreg.0002 # grant secret_homosexual reliably to a same-sex couple
		mpreg.0004 # start the independent conception chain if either partner is male (Lesson 32)
	}
}

# common/on_action/relations/relation_on_actions.txt:45 -- fires when the lover
# relation is set. root = the character who set it, scope:target = the new lover.
on_set_relation_lover = {
	events = {
		mpreg.0005 # start the independent conception chain if either partner is male (Lesson 32)
	}
}
